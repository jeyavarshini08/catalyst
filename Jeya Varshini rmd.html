# dataset loading
finaldata <- read.csv("C:\\Users\\jeyav\\Downloads\\Catalyst assignment\\finaldata.csv")

unique(finaldata$District)
## [1] "Hyderabad " "Vikarabad"
# Trimming 
finaldata$District <- trimws(finaldata$District)

# filtering trims 
hyderabad <- finaldata[finaldata$District == "Hyderabad", ]

# Filtering data for hyd
hyderabad <- subset(finaldata, District == "Hyderabad")

# Reshaping
library(tidyr)
## Warning: package 'tidyr' was built under R version 4.3.3
hyderabad_long <- hyderabad %>%
  pivot_longer(cols = c(Avg_haemoglobin_women, Avg_Temp),
               names_to = "Variable", 
               values_to = "Value")
# Plot the grouped bar chart
library(ggplot2)

ggplot(hyderabad_long, aes(x = Month, y = Value, fill = Variable)) +
  geom_bar(stat = "identity", position = "dodge", width = 0.5) +
  geom_text(aes(label = round(Value, 1)), 
            position = position_dodge(width = 0.5), 
            vjust = -0.5, size = 3) +  # Label placement and text size
  labs(title = "Female Haemoglobin lvl vs Temperatures in May & Dec - Hyderabad",
       x = "Month",
       y = "Value") +
  scale_fill_manual(values = c("red3", "yellow2"), 
                    labels = c("Haemoglobin (g/dL)", "Temperature (°C)")) +
  theme_minimal()
